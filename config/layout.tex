% การตั้งค่ารูปแบบเอกสารส่วนกลาง / Shared document layout configuration

% ภาษาไทย ยูนิโคด และฟอนต์ / Thai language, Unicode, and fonts
\usepackage{fontspec}
\usepackage{polyglossia}
\setdefaultlanguage{thai}
\setotherlanguage{english}
\XeTeXlinebreaklocale "th"
\XeTeXlinebreakskip=0pt plus 1pt

\setmainfont[
  Path=fonts/,
  UprightFont={THSarabunNew.ttf},
  BoldFont={THSarabunNew Bold.ttf},
  ItalicFont={THSarabunNew Italic.ttf},
  BoldItalicFont={THSarabunNew BoldItalic.ttf}
]{THSarabunNew}

% หน้ากระดาษและย่อหน้า / Page geometry and paragraphs
\usepackage{geometry}
\geometry{top=3.81cm,bottom=2.54cm,left=3.81cm,right=2.54cm}
\usepackage{setspace}
\singlespacing
\usepackage{indentfirst}
\AtBeginDocument{\fontsize{16pt}{16pt}\selectfont}
\setlength{\parindent}{1.27cm}
\setlength{\parskip}{0pt}

% หัวบทและหัวข้อ / Chapter and heading styles
\usepackage[indentafter]{titlesec}
\newlength{\chapterBeforeSpace}
\newlength{\chapterAfterSpace}
\setlength{\chapterBeforeSpace}{0pt}
\setlength{\chapterAfterSpace}{16pt}
\titleformat{\chapter}[block]
  {\centering\bfseries\fontsize{18pt}{22pt}\selectfont}
  {}{0pt}{\chaptertitlename\ \thechapter\protect\\}
\titleformat{name=\chapter,numberless}[block]
  {\centering\bfseries\fontsize{18pt}{22pt}\selectfont}
  {}{0pt}{}
\titlespacing*{\chapter}{0pt}{\chapterBeforeSpace}{\chapterAfterSpace}
\titlespacing*{name=\chapter,numberless}{0pt}{\chapterBeforeSpace}{\chapterAfterSpace}

% แสดงเลขหัวข้อถึง 1.1.1.1 แต่สารบัญแสดงถึง 1.1 / Number through 1.1.1.1; list through 1.1 in the TOC
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{1}
\titleformat{\section}
  {\bfseries\fontsize{16pt}{16pt}\selectfont}
  {\thesection}{1em}{}
\titleformat{\subsection}
  {\normalfont\fontsize{16pt}{16pt}\selectfont}
  {\thesubsection}{1em}{}
\titleformat{\subsubsection}
  {\normalfont\fontsize{16pt}{16pt}\selectfont}
  {\thesubsubsection}{1em}{}
\titlespacing*{\section}{0pt}{12pt}{0pt}
\titlespacing*{\subsection}{0.95cm}{0pt}{0pt}
\titlespacing*{\subsubsection}{2.22cm}{0pt}{0pt}

% รูปภาพ ตาราง รายการ และสมการ / Figures, tables, lists, and equations
\usepackage{graphicx}
\graphicspath{{figures/}}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{tabularx}
\usepackage{enumitem}
\setlist{nosep}
\usepackage{amsmath,amssymb}

% คำบรรยายและเลขลำดับแบบเลขบท-ลำดับ / Captions and chapter-sequence numbering
\usepackage{caption}
\renewcommand{\thetable}{\thechapter-\arabic{table}}
\renewcommand{\thefigure}{\thechapter-\arabic{figure}}
\renewcommand{\tablename}{ตารางที่}
\renewcommand{\figurename}{ภาพที่}
\renewcommand{\listtablename}{สารบัญตาราง}
\renewcommand{\listfigurename}{สารบัญภาพ}
\captionsetup[table]{position=top,justification=raggedright,singlelinecheck=false,
  labelfont=bf,textfont=normalfont,labelsep=space,skip=6pt}
\captionsetup[figure]{position=bottom,justification=centering,
  labelfont=bf,textfont=normalfont,labelsep=space,skip=6pt}

% เลขหน้าส่วนเนื้อหาที่มุมขวาบน / Main-matter page numbers at the top right
\usepackage{fancyhdr}
\fancypagestyle{crubody}{%
  \fancyhf{}%
  \fancyhead[R]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}
